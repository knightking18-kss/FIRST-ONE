```javascript
// script.js
const ordersForm = document.getElementById('orders-form');
const customerSelect = document.getElementById('customer');
const orderDateInput = document.getElementById('order-date');
const ordersTable = document.getElementById('orders-table');

ordersForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const customer = customerSelect.value;
        const orderDate = orderDateInput.value;

        if (customer && orderDate) {
                // Send place order request to backend API
                fetch('/api/orders', {
                        method: 'POST',
                        headers: {
                                'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({ customer, orderDate }),
                })
                        .then((response) => response.json())
                        .then((data) => {
                                // Order placed successfully, display success message
                                alert('Order placed successfully');
                                // Fetch and display updated order data
                                fetch('/api/orders')
                                        .then((response) => response.json())
                                        .then((data) => {
                                                // Clear existing order data
                                                const ordersTableBody = ordersTable.tBodies[0];
                                                while (ordersTableBody.rows.length > 0) {
                                                        ordersTableBody.deleteRow(0);
                                                }
                                                // Display updated order data
                                                data.forEach((item) => {
                                                        const row = ordersTable.insertRow(-1);
                                                        const orderIDCell = row.insertCell(0);
                                                        const customerCell = row.insertCell(1);
                                                        const orderDateCell = row.insertCell(2);
                                                        const orderStatusCell = row.insertCell(3);
                                                        orderIDCell.textContent = item.orderID;
                                                        customerCell.textContent = item.customer;
                                                        orderDateCell.textContent = item.orderDate;
                                                        orderStatusCell.textContent = item.orderStatus;
                                                });
                                        })
                                        .catch((error) => {
                                                // Error fetching order data
                                                alert('Error fetching order data');
                                        });
                        })
                        .catch((error) => {
                                // Error placing order
                                alert('Error placing order');
                        });
        } else {
                // Invalid input
                alert('Please enter valid input');
        }
});
```
