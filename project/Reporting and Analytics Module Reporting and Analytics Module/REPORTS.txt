```javascript
// script.js
const reportsForm = document.getElementById('reports-form');
const reportTypeSelect = document.getElementById('report-type');
const reportsTable = document.getElementById('reports-table');

reportsForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const reportType = reportTypeSelect.value;

        if (reportType) {
                // Send generate report request to backend API
                fetch('/api/reports', {
                        method: 'POST',
                        headers: {
                                'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({ reportType }),
                })
                        .then((response) => response.json())
                        .then((data) => {
                                // Report generated successfully, display success message
                                alert('Report generated successfully');
                                // Fetch and display updated report data
                                fetch('/api/reports')
                                        .then((response) => response.json())
                                        .then((data) => {
                                                // Clear existing report data
                                                const reportsTableBody = reportsTable.tBodies[0];
                                                while (reportsTableBody.rows.length > 0) {
                                                        reportsTableBody.deleteRow(0);
                                                }
                                                // Display updated report data
                                                data.forEach((item) => {
                                                        const row = reportsTable.insertRow(-1);
                                                        const reportIDCell = row.insertCell(0);
                                                        const reportTypeCell = row.insertCell(1);
                                                        const reportDateCell = row.insertCell(2);
                                                        reportIDCell.textContent = item.reportID;
                                                        reportTypeCell.textContent = item.reportType;
                                                        reportDateCell.textContent = item.reportDate;
                                                });
                                        })
                                        .catch((error) => {
                                                // Error fetching report data
                                                alert('Error fetching report data');
                                        });
                        })
                        .catch((error) => {
                                // Error generating report
                                alert('Error generating report');
                        });
        } else {
                // Invalid input
                alert('Please enter valid input');
        }
});
```
